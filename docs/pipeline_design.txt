class Resource {
	// nothing
};

enum class TextureType {
	Texture1D,
	Texture2D,
	Texture3D,
	Cubemap,
};

class Texture : public Resource {
	TextureType type;
};

enum class BufferType {
	Vertex,
	Index,
	Uniform,
};

class Buffer : public Resource {
	BufferType type;
};

enum class ShaderType {
	Vertex,
	Fragment,
	Compute,
};

class ShaderInterface {
	std::vector<ShaderInterfaceVariable> inputs;
	std::vector<ShaderInterfaceVariable> outputs;
	std::vector<ShaderResource> resources;

	// will check that the left outputs match with the right inputs, and resources don't overlap 
	//   if a discrepancy is detected, should be a runtime error 
	// the generated shader interface will have the left inputs, right outputs, and both resources 
	static ShaderInterface unify(ShaderInterface& left, ShaderInterface& right);
};

class Shader {
	ShaderType type;
	ShaderInterface interface;
};


class ProgramType {
	Rendering,	// vertex + fragment shader
	Compute,	// compute shader
};

// a Program should be a container for the code (shaders) required to execute the action 
class Program {
protected:
	// unified interface of all shaders of this program 
	ShaderInterface interface;

public:
    // should make sure that the sequence of shaders actually form a valid program 
    virtual void validate() = 0;
};

class RenderingProgram : public Program {
public:
	// generate invocation 
	virtual RenderingProgramInvocation generate_invocation() = 0;
};

class OpenGLRenderingProgram : public RenderingProgram {
private:
	// stuff related to underlying graphics API 
	OpenGLGraphicsDevice& device;
	GLuint handle;

public:
	// generate invocation 
	OpenGLRenderingProgramInvocation generate_invocation() override;
};

// a ProgramInvocation is a container relating a Program and the resources required to run the program 
// is it useful to have all the different types of program invocations inherit from a base class?
class ProgramInvocation {
protected:
	// bound resources to all shaders in this program
	struct ResourceBinding {
		Resource& resource;
		uint32_t set;
		uint32_t binding;
	};
	std::vector<ResourceBinding> resources;

public:
    // should make sure that the resources contained here can actually be used to invoke the given program
    // should only be able to invoke after validating 
    // once validated, you won't be able to change any resources 
    virtual void validate() = 0;
};

// everything required to invoke a rendering program
class RenderingProgramInvocation : public ProgramInvocation {
protected:
	// how to assign each element in the bound vertex buffer
	enum class InputRate {
		Vertex,		// every element goes to a different vertex
		Instance,	// every element goes to a different instance
	};

    // how each vertex attribute element should be interpreted
	enum class VertexFormat {
		Float32x1,
        Float32x2,
        Float32x3,
        Float32x4,
	};

    // how each index element should be interpreted
	enum class IndexFormat {
        UInt16,
		UInt32,
	};

	// mapping a vertex buffer to an input interface variable
	struct VertexAttributeBinding {
		Buffer& buffer;         // what buffer to use 
		uint32_t location;      // what interface variable to bind to 
		uint32_t offset;        // byte of the first element
		uint32_t stride;        // bytes between the start of each element
		InputRate rate;         // how to map elements
		VertexFormat format;    // how to interpret elements
	};
	std::vector<VertexBufferBinding> attribute_bindings;

	// optionally bind an index buffer for indexed rendering
	struct IndexBufferBinding {
		Buffer& buffer;
		IndexFormat format;
	};
	std::optional<Buffer&> index_buffer;

    // mapping a texture to an output interface variable
    struct TextureOutputBinding {
        Texture& texture;       // what texture to use 
        uint32_t location;
    };
    std::vector<TextureOutputBinding> output_textures;

public:
	// invoke this program 
	virtual void draw(uint32_t nr_elements, uint32_t nr_instances) = 0;
};

class OpenGLRenderingProgramInvocation : public RenderingProgramInvocation {
private:
	// underlying graphics API 
	OpenGLGraphicsDevice& device;

	// underlying program 
	OpenGLRenderingProgram& program;

public:
	// invoke this program 
	void draw(uint32_t nr_elements, uint32_t nr_instances) override;
};
